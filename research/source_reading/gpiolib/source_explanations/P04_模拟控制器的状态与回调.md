# 第4章\_模拟控制器的状态与回调

gpio-sim让真实gpiolib请求作用在内存位图上。它适合检查所有权、方向和值的交接，却不能证明真实pad的电气行为。本页唯一展开 `drivers/gpio/gpio-sim.c` 中本轮使用的状态与回调；先读 [模拟器模块导读](../navigation/P03_GPIO模拟器如何兑现控制器能力.md#3.1_从Consumer的三根线追到一个bank)，固定提交与配置边界见 [总索引](../navigation/P01_Linux_6.12_GPIO源码阅读索引.md#1.1_版本身份与阅读路线)。下方Doxygen中文说明是仓库补充，源码摘录用于阅读，不是可独立编译的驱动。

原件为 [NXP固定提交中的gpio-sim.c](https://github.com/nxp-imx/linux-imx/blob/dfaf2136deb2af2e60b994421281ba42f1c087e0/drivers/gpio/gpio-sim.c)，文件沿用上游 `GPL-2.0-or-later` 许可及作者声明。与本页相关的完整驱动、configfs、sysfs和IRQ辅助实现均保留在原件；摘录没有另造省略函数的替代定义。

## 4.1\_位图怎样承接每线状态

一个bank可提供多根线，不能只用一个bool同时表示“被请求”“输入”“值为高”。实现把这几组状态分开，并把所有位图放在一个私有对象中。

```c
/**
 * @brief 一个bank的私有状态；指针指向按线路数分配的位图。
 * @note 仓库补充的中文阅读说明；下方函数或结构保留固定版本原文。
 */
struct gpio_sim_chip {
	struct gpio_chip gc;
	struct device *dev;
	unsigned long *request_map;
	unsigned long *direction_map;
	unsigned long *value_map;
	unsigned long *pull_map;
	struct irq_domain *irq_sim;
	struct mutex lock;
	const struct attribute_group **attr_groups;
};
```

`gc` 嵌入公共能力；回调通过 `gpiochip_get_data(gc)` 取得这个外层对象。`dev` 在登记后指向找到的gpiochip子设备，用于观察接口。四个位图的bit索引都是该bank内offset：request记录请求，direction的1为输入，value保存当前模拟值，pull保存模拟pull选择。它们不包含公共desc标签或gdev引用。

`irq_sim` 是模拟中断映射域，保存本地线路接入IRQ体系的关系；`lock` 是保护相关位图访问的mutex；`attr_groups` 是sysfs观察属性组，不是硬件状态。位图的具体存储由后面的add_bank分配，所以仅分配外层结构尚不足以调用回调。

## 4.2\_锁怎样保护读写方向与批量更新

这些回调没有MMIO读写。它们先从gc取得实例，再在该实例的mutex下访问位图。同一个锁把Consumer操作与模拟输入注入协调起来。

```c
/**
 * @brief 单线和批量访问共享同一把锁；set没有错误返回通道。
 * @note 仓库补充的中文阅读说明；下方函数或结构保留固定版本原文。
 */
static int gpio_sim_get(struct gpio_chip *gc, unsigned int offset)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	guard(mutex)(&chip->lock);

	return !!test_bit(offset, chip->value_map);
}

static void gpio_sim_set(struct gpio_chip *gc, unsigned int offset, int value)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	scoped_guard(mutex, &chip->lock)
		__assign_bit(offset, chip->value_map, value);
}

static int gpio_sim_get_multiple(struct gpio_chip *gc,
				 unsigned long *mask, unsigned long *bits)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	scoped_guard(mutex, &chip->lock)
		bitmap_replace(bits, bits, chip->value_map, mask, gc->ngpio);

	return 0;
}

static void gpio_sim_set_multiple(struct gpio_chip *gc,
				  unsigned long *mask, unsigned long *bits)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	scoped_guard(mutex, &chip->lock)
		bitmap_replace(chip->value_map, chip->value_map, bits, mask,
			       gc->ngpio);
}
```

`guard(mutex)` 在当前作用域退出时解锁，`scoped_guard` 把锁限定在紧随其后的语句或代码块。`!!test_bit()` 把测试结果归一化为0/1。`__assign_bit` 本身不是用来替代锁的跨CPU完整协议，这里由mutex保护；函数返回以后锁已放下。

`bitmap_replace(dst, old, new, mask, ngpio)` 保留mask未选中的old位，只把选中位替换为new位。get_multiple把value_map中被选中的位写入调用者bits，set_multiple则把调用者bits合并到value_map；整个合并在一次锁区间内完成。把它改成逐位锁定，可能保持最终值，却失去一次事务的观察边界。

方向回调继续使用同一保护方式：

```c
/**
 * @brief 方向与初值的模拟事务；方向位1表示输入。
 * @note 仓库补充的中文阅读说明；下方函数或结构保留固定版本原文。
 */
static int gpio_sim_direction_output(struct gpio_chip *gc,
				     unsigned int offset, int value)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	scoped_guard(mutex, &chip->lock) {
		__clear_bit(offset, chip->direction_map);
		__assign_bit(offset, chip->value_map, value);
	}

	return 0;
}

static int gpio_sim_direction_input(struct gpio_chip *gc, unsigned int offset)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	scoped_guard(mutex, &chip->lock)
		__set_bit(offset, chip->direction_map);

	return 0;
}

static int gpio_sim_get_direction(struct gpio_chip *gc, unsigned int offset)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);
	int direction;

	scoped_guard(mutex, &chip->lock)
		direction = !!test_bit(offset, chip->direction_map);

	return direction ? GPIO_LINE_DIRECTION_IN : GPIO_LINE_DIRECTION_OUT;
}
```

方向输出先清direction位，再写value，但两步都在同一个mutex区间里，使用同一锁的观察者不能插入中间。方向输入只置direction位，不在此处把value重置为pull。查询方向读取该位并返回GPIO方向常量：输入1、输出0。不能把“设为输入”自行扩写成源码没有执行的“立即重新采样外部电平”。

模拟器没有真实输出使能电路，因此上述语句次序不能作为真实MMIO控制器避免毛刺的硬件证据。它证明的是锁保护的内存状态交接。实际Provider方向/值回调的返回契约见 [gpio_chip成员定义](https://github.com/nxp-imx/linux-imx/blob/dfaf2136deb2af2e60b994421281ba42f1c087e0/include/linux/gpio/driver.h#L323)。

## 4.3\_为什么释放后读到的值可能改变

公共层先取得线路占用，再调用模拟器request；释放时调用free。这里的请求位为模拟行为服务，并不重复实现公共层的排他判定。

```c
/**
 * @brief 释放用pull值替换当前value，和公共占用撤销是两件事。
 * @note 仓库补充的中文阅读说明；下方函数或结构保留固定版本原文。
 */
static int gpio_sim_request(struct gpio_chip *gc, unsigned int offset)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	scoped_guard(mutex, &chip->lock)
		__set_bit(offset, chip->request_map);

	return 0;
}

static void gpio_sim_free(struct gpio_chip *gc, unsigned int offset)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	scoped_guard(mutex, &chip->lock) {
		__assign_bit(offset, chip->value_map,
			     !!test_bit(offset, chip->pull_map));
		__clear_bit(offset, chip->request_map);
	}
}
```

request只设请求位；free先把value改为pull再清请求位，两步受同一个锁保护。free没有改变direction位。所以释放后的值和方向不能只凭“请求位已清”推断；这也是Consumer停机写入某值后，最终观察可能不同的具体原因。

pull又如何改变？模拟器配置和实验注入最终会经过下面的函数。它区分“仍被请求的输入”“仍被请求的输出”和“无人请求”，而不是把任何pull写入都直接当成电平变化。

```c
/**
 * @brief 按请求和方向状态传播模拟pull，并在符合条件时产生模拟IRQ。
 * @note 仓库补充的中文阅读说明；下方函数或结构保留固定版本原文。
 */
static int gpio_sim_apply_pull(struct gpio_sim_chip *chip,
			       unsigned int offset, int value)
{
	int irq, irq_type, ret;

	guard(mutex)(&chip->lock);

	if (test_bit(offset, chip->request_map) &&
	    test_bit(offset, chip->direction_map)) {
		if (value == !!test_bit(offset, chip->value_map))
			goto set_pull;

		/*
		 * This is fine - it just means, nobody is listening
		 * for interrupts on this line, otherwise
		 * irq_create_mapping() would have been called from
		 * the to_irq() callback.
		 */
		irq = irq_find_mapping(chip->irq_sim, offset);
		if (!irq)
			goto set_value;

		irq_type = irq_get_trigger_type(irq);

		if ((value && (irq_type & IRQ_TYPE_EDGE_RISING)) ||
		    (!value && (irq_type & IRQ_TYPE_EDGE_FALLING))) {
			ret = irq_set_irqchip_state(irq, IRQCHIP_STATE_PENDING,
						    true);
			if (ret)
				goto set_pull;
		}
	}

set_value:
	/* Change the value unless we're actively driving the line. */
	if (!test_bit(offset, chip->request_map) ||
	    test_bit(offset, chip->direction_map))
		__assign_bit(offset, chip->value_map, value);

set_pull:
	__assign_bit(offset, chip->pull_map, value);
	return 0;
}
```

先看普通值传播：`set_value` 只有在未被请求或当前为输入时才更新value；仍被请求的输出保留主动驱动值。`set_pull` 无论是否更新value都会保存新的pull值，供后续free等路径使用。对仍被请求的输入，如果新值与现值相同，直接更新pull而不再产生变化事件。

再看IRQ分支：输入确有变化时，`irq_find_mapping` 只查已有映射；没有映射就直接改值，已有映射也不等于已经安装handler。已有映射时按 `irq_get_trigger_type` 的上升/下降沿条件决定是否通过 `irq_set_irqchip_state(..., IRQCHIP_STATE_PENDING, true)` 通知模拟域。若该调用报错，源码跳到set_pull，value不经过set_value更新，但函数最终仍返回0。这是此版本实现的实际边界，不能概括为“所有模拟输入失败都返回调用者”。

配置参数和映射入口如下：

```c
/**
 * @brief 仅支持两种bias配置；to_irq可以创建映射而非只查询。
 * @note 仓库补充的中文阅读说明；下方函数或结构保留固定版本原文。
 */
static int gpio_sim_set_config(struct gpio_chip *gc, unsigned int offset,
			       unsigned long config)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	switch (pinconf_to_config_param(config)) {
	case PIN_CONFIG_BIAS_PULL_UP:
		return gpio_sim_apply_pull(chip, offset, 1);
	case PIN_CONFIG_BIAS_PULL_DOWN:
		return gpio_sim_apply_pull(chip, offset, 0);
	default:
		break;
	}

	return -ENOTSUPP;
}

static int gpio_sim_to_irq(struct gpio_chip *gc, unsigned int offset)
{
	struct gpio_sim_chip *chip = gpiochip_get_data(gc);

	return irq_create_mapping(chip->irq_sim, offset);
}
```

`pinconf_to_config_param` 从打包配置中取参数类型，本实现只处理上拉和下拉，其余返回 `-ENOTSUPP`。`gpio_sim_to_irq` 使用 `irq_create_mapping`，因此Consumer调用gpiod_to_irq可能推动映射建立，不是纯查询；它没有在此处配置输入方向或安装处理者。request/free和set_config可以使用可睡眠路径，to_irq仍须满足公共接口规定的不可睡眠约束。

事件身份与后续处理者的协作进入 [GPIO中断主线](../../../../knowledge/driver_model/gpio/P07_GPIO_中断桥接与事件传播.md)；本节只闭合模拟输入如何改变状态并交给事件域。

## 4.4\_怎样在发布回调前准备好它们依赖的状态

S1登记不能先发布空壳gc，再慢慢给回调补锁和位图。add_bank先从固件节点读取线路数和标签，再准备全部基本状态，最后登记。

```c
/**
 * @brief 按bank属性准备私有状态，登记后再建立观察接口；错误交回probe清理。
 * @note 仓库补充的中文阅读说明；下方函数或结构保留固定版本原文。
 */
static int gpio_sim_add_bank(struct fwnode_handle *swnode, struct device *dev)
{
	struct gpio_sim_chip *chip;
	struct gpio_chip *gc;
	const char *label;
	u32 num_lines;
	int ret;

	/* 仓库补充：从当前bank取得能力数量，不能套用固定32线。 */
	ret = fwnode_property_read_u32(swnode, "ngpios", &num_lines);
	if (ret)
		return ret;

	if (num_lines > GPIO_SIM_NGPIO_MAX)
		return -ERANGE;

	ret = fwnode_property_read_string(swnode, "gpio-sim,label", &label);
	if (ret) {
		label = devm_kasprintf(dev, GFP_KERNEL, "%s:%pfwP",
				       dev_name(dev), swnode);
		if (!label)
			return -ENOMEM;
	}

	chip = devm_kzalloc(dev, sizeof(*chip), GFP_KERNEL);
	if (!chip)
		return -ENOMEM;

	/* 仓库补充：各位图分别分配，外层对象存在还不等于回调可运行。 */
	chip->request_map = devm_bitmap_zalloc(dev, num_lines, GFP_KERNEL);
	if (!chip->request_map)
		return -ENOMEM;

	chip->direction_map = devm_bitmap_alloc(dev, num_lines, GFP_KERNEL);
	if (!chip->direction_map)
		return -ENOMEM;

	/* Default to input mode. */
	bitmap_fill(chip->direction_map, num_lines);

	chip->value_map = devm_bitmap_zalloc(dev, num_lines, GFP_KERNEL);
	if (!chip->value_map)
		return -ENOMEM;

	chip->pull_map = devm_bitmap_zalloc(dev, num_lines, GFP_KERNEL);
	if (!chip->pull_map)
		return -ENOMEM;

	chip->irq_sim = devm_irq_domain_create_sim_full(dev, swnode, num_lines,
							&gpio_sim_irq_sim_ops,
							chip);
	if (IS_ERR(chip->irq_sim))
		return PTR_ERR(chip->irq_sim);

	ret = devm_add_action_or_reset(dev, gpio_sim_dispose_mappings, chip);
	if (ret)
		return ret;

	ret = devm_mutex_init(dev, &chip->lock);
	if (ret)
		return ret;

	/* 仓库补充：锁和模拟IRQ域就绪后，填写将要发布的能力。 */
	gc = &chip->gc;
	gc->base = -1;
	gc->ngpio = num_lines;
	gc->label = label;
	gc->owner = THIS_MODULE;
	gc->parent = dev;
	gc->fwnode = swnode;
	gc->get = gpio_sim_get;
	gc->set = gpio_sim_set;
	gc->get_multiple = gpio_sim_get_multiple;
	gc->set_multiple = gpio_sim_set_multiple;
	gc->direction_output = gpio_sim_direction_output;
	gc->direction_input = gpio_sim_direction_input;
	gc->get_direction = gpio_sim_get_direction;
	gc->set_config = gpio_sim_set_config;
	gc->to_irq = gpio_sim_to_irq;
	gc->request = gpio_sim_request;
	gc->free = gpio_sim_free;
	gc->dbg_show = PTR_IF(IS_ENABLED(CONFIG_DEBUG_FS), gpio_sim_dbg_show);
	gc->can_sleep = true;

	/* 仓库补充：此后公共层可以调用这些回调。 */
	ret = devm_gpiochip_add_data(dev, gc, chip);
	if (ret)
		return ret;

	/* 仓库补充：后续观察接口若失败，已经登记的能力也需要撤销。 */
	chip->dev = device_find_child(dev, swnode, gpio_sim_dev_match_fwnode);
	if (!chip->dev)
		return -ENODEV;

	ret = devm_add_action_or_reset(dev, gpio_sim_put_device, chip->dev);
	if (ret)
		return ret;

	/* Used by sysfs callbacks. */
	dev_set_drvdata(chip->dev, chip);

	return gpio_sim_setup_sysfs(chip);
}
```

这段完整函数可以分四步阅读。首先 `fwnode_property_read_u32` 取得ngpios，超过本文件上限1024时报 `-ERANGE`；属性缺失时直接返回读取错误。可选标签缺失时以设备与固件路径生成标签。这里的固件节点可来自configfs创建的software node，不要求一定来自设备树。

其次分配外层对象和四个位图：request、value、pull清零，direction填1，建立默认输入状态。模拟IRQ域返回错误指针时用 `PTR_ERR` 传回；映射清理动作与mutex都在gc注册之前就绪。`devm_add_action_or_reset` 登记动作失败时会立即执行该动作，不能把失败理解成什么也没发生。

第三步填写gc。`base=-1` 请求动态分配旧整数空间的base；ngpio是已读取的线路数；parent与fwnode分别指出设备和bank身份；各函数指针把本页回调接入公共层。`PTR_IF(IS_ENABLED(CONFIG_DEBUG_FS), ...)` 只在相应配置启用时提供调试显示函数。因为回调使用mutex，can_sleep为true。`devm_gpiochip_add_data` 才把能力和私有data交给公共层。

第四步在登记后找gpiochip子设备，保存引用并登记put动作，将私有对象挂到该设备以供sysfs回调使用，最后建立观察接口。若后半段失败，已经登记的gpiochip也需要随probe失败清理，不能只释放最后分配的一块内存。其辅助函数的职责按调用参数可分清：`gpio_sim_dispose_mappings` 处理映射退出，`gpio_sim_put_device` 放设备引用，`gpio_sim_setup_sysfs` 建立每线观察入口；本页不重复展开这些与值事务无关的函数体。

## 4.5\_多个bank怎样由同一个probe收束

设备可以有多个bank，probe按子节点逐个建立。失败时不仅要返回错误，还要放下迭代中当前节点的引用。

```c
/**
 * @brief 遍历bank并传播第一个错误；设备资源清理覆盖已经建立的bank。
 * @note 仓库补充的中文阅读说明；下方函数或结构保留固定版本原文。
 */
static int gpio_sim_probe(struct platform_device *pdev)
{
	struct device *dev = &pdev->dev;
	struct fwnode_handle *swnode;
	int ret;

	device_for_each_child_node(dev, swnode) {
		ret = gpio_sim_add_bank(swnode, dev);
		if (ret) {
			fwnode_handle_put(swnode);
			return ret;
		}
	}

	return 0;
}
```

成功路径继续下一个子节点；失败路径显式 `fwnode_handle_put(swnode)` 再返回ret，后续由驱动核心执行probe失败的devres清理。该循环不表示多bank登记从未对外可见，也不负责替业务Consumer停机。

验证理解时可预测三种变化：删除set的锁会失去与pull注入共享状态的同步；改变free的value赋值会改变释放后观察结果而不必改变公共desc占用；把ngpio改成一个猜测常数会让公共层暴露的线路集合与实例配置不符。它们分别破坏同步、模拟语义和能力身份，不能都归为“GPIO读写错误”。

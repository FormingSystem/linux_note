# 第6章\_用户请求的发布与回收

本页追踪一次v2请求从分配到最后释放，重点是资源何时被记入清理范围，以及成功交付与硬件动作之间的区别。先读 [用户请求模块导读](../navigation/P05_用户请求怎样持有线路与事件.md#5.1_先从两个文件对象区分入口和持有者)，版本与配置见 [总索引](../navigation/P01_Linux_6.12_GPIO源码阅读索引.md#1.1_版本身份与阅读路线)。

所有摘录来自NXP Linux 6.12.20固定提交 `dfaf2136deb2af2e60b994421281ba42f1c087e0` 的 `drivers/gpio/gpiolib-cdev.c`，沿用GPL-2.0许可。中文Doxygen是仓库补充的阅读说明，原函数体按该版本保留，不作为用户程序复制编译。

## 6.1\_共享请求状态怎样连接多根线路

每条线需要自己的描述符、IRQ和边沿状态，整个请求则需要统一的文件寿命、配置串行化和事件出口。linereq把共享状态放在固定头部，尾部lines按请求数量分配：

```c
/**
 * @brief 保存请求共享状态；lines柔性数组与请求对象一起分配。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
struct linereq {
	struct gpio_device *gdev;
	const char *label;
	u32 num_lines;
	wait_queue_head_t wait;
	struct notifier_block device_unregistered_nb;
	u32 event_buffer_size;
	DECLARE_KFIFO_PTR(events, struct gpio_v2_line_event);
	atomic_t seqno;
	struct mutex config_mutex;
	struct line lines[] __counted_by(num_lines);
};
```

`gdev`保护控制器对象寿命，`label`提供诊断名称，`num_lines`限定尾部数组。`config_mutex`串行化相应ioctl多步配置；`wait`既是阻塞读等待队列，其内嵌lock也保护事件FIFO。`events`是动态kfifo描述，`event_buffer_size`保存建议容量；声明和INIT_KFIFO本身都不分配实际事件存储。`seqno`为多线请求产生请求级序号，单线请求直接使用每线序号。

每个line保存desc并通过req回指这个共享对象，还保存irq、edflags、timestamp_ns、req_seqno、line_seqno、去抖work等运行状态。生产者不能只保存desc而丢掉请求归属，否则无法知道该唤醒哪个读者、把事件放入哪个FIFO。`device_unregistered_nb`用于接收gdev注销通知，不把对象引用变成永久硬件可用保证。

## 6.2\_为什么先配置线路再发布文件

下面完整保留创建函数，因为其早期错误、逐线失败、文件已建立后的失败各有不同清理条件。统一的错误出口不是把所有阶段都机械执行两次free。

```c
/**
 * @brief 先记录每项已取得资源，再在全部线路建立后发布fd；文件接管后避免重复清理。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static int linereq_create(struct gpio_device *gdev, void __user *ip)
{
	struct gpio_v2_line_request ulr;
	struct gpio_v2_line_config *lc;
	struct linereq *lr;
	struct file *file;
	u64 flags, edflags;
	unsigned int i;
	int fd, ret;

	if (copy_from_user(&ulr, ip, sizeof(ulr)))
		return -EFAULT;

	if ((ulr.num_lines == 0) || (ulr.num_lines > GPIO_V2_LINES_MAX))
		return -EINVAL;

	if (memchr_inv(ulr.padding, 0, sizeof(ulr.padding)))
		return -EINVAL;

	lc = &ulr.config;
	ret = gpio_v2_line_config_validate(lc, ulr.num_lines);
	if (ret)
		return ret;

	lr = kvzalloc(struct_size(lr, lines, ulr.num_lines), GFP_KERNEL);
	if (!lr)
		return -ENOMEM;
	lr->num_lines = ulr.num_lines;

	lr->gdev = gpio_device_get(gdev);

	for (i = 0; i < ulr.num_lines; i++) {
		lr->lines[i].req = lr;
		WRITE_ONCE(lr->lines[i].sw_debounced, 0);
		INIT_DELAYED_WORK(&lr->lines[i].work, debounce_work_func);
	}

	if (ulr.consumer[0] != '\0') {
		/* label is only initialized if consumer is set */
		lr->label = kstrndup(ulr.consumer, sizeof(ulr.consumer) - 1,
				     GFP_KERNEL);
		if (!lr->label) {
			ret = -ENOMEM;
			goto out_free_linereq;
		}
	}

	mutex_init(&lr->config_mutex);
	init_waitqueue_head(&lr->wait);
	INIT_KFIFO(lr->events);
	lr->event_buffer_size = ulr.event_buffer_size;
	if (lr->event_buffer_size == 0)
		lr->event_buffer_size = ulr.num_lines * 16;
	else if (lr->event_buffer_size > GPIO_V2_LINES_MAX * 16)
		lr->event_buffer_size = GPIO_V2_LINES_MAX * 16;

	atomic_set(&lr->seqno, 0);

	/* Request each GPIO */
	for (i = 0; i < ulr.num_lines; i++) {
		u32 offset = ulr.offsets[i];
		struct gpio_desc *desc = gpio_device_get_desc(gdev, offset);

		if (IS_ERR(desc)) {
			ret = PTR_ERR(desc);
			goto out_free_linereq;
		}

		ret = gpiod_request_user(desc, lr->label);
		if (ret)
			goto out_free_linereq;

		lr->lines[i].desc = desc;
		flags = gpio_v2_line_config_flags(lc, i);
		gpio_v2_line_config_flags_to_desc_flags(flags, &desc->flags);

		ret = gpiod_set_transitory(desc, false);
		if (ret < 0)
			goto out_free_linereq;

		edflags = flags & GPIO_V2_LINE_EDGE_DETECTOR_FLAGS;
		/*
		 * Lines have to be requested explicitly for input
		 * or output, else the line will be treated "as is".
		 */
		if (flags & GPIO_V2_LINE_FLAG_OUTPUT) {
			int val = gpio_v2_line_config_output_value(lc, i);

			ret = gpiod_direction_output(desc, val);
			if (ret)
				goto out_free_linereq;
		} else if (flags & GPIO_V2_LINE_FLAG_INPUT) {
			ret = gpiod_direction_input(desc);
			if (ret)
				goto out_free_linereq;

			ret = edge_detector_setup(&lr->lines[i], lc, i,
						  edflags);
			if (ret)
				goto out_free_linereq;
		}

		lr->lines[i].edflags = edflags;

		gpiod_line_state_notify(desc, GPIO_V2_LINE_CHANGED_REQUESTED);

		dev_dbg(&gdev->dev, "registered chardev handle for line %d\n",
			offset);
	}

	lr->device_unregistered_nb.notifier_call = linereq_unregistered_notify;
	ret = blocking_notifier_chain_register(&gdev->device_notifier,
					       &lr->device_unregistered_nb);
	if (ret)
		goto out_free_linereq;

	fd = get_unused_fd_flags(O_RDONLY | O_CLOEXEC);
	if (fd < 0) {
		ret = fd;
		goto out_free_linereq;
	}

	file = anon_inode_getfile("gpio-line", &line_fileops, lr,
				  O_RDONLY | O_CLOEXEC);
	if (IS_ERR(file)) {
		ret = PTR_ERR(file);
		goto out_put_unused_fd;
	}

	ulr.fd = fd;
	if (copy_to_user(ip, &ulr, sizeof(ulr))) {
		/*
		 * fput() will trigger the release() callback, so do not go onto
		 * the regular error cleanup path here.
		 */
		fput(file);
		put_unused_fd(fd);
		return -EFAULT;
	}

	fd_install(fd, file);

	dev_dbg(&gdev->dev, "registered chardev handle for %d lines\n",
		lr->num_lines);

	return 0;

out_put_unused_fd:
	put_unused_fd(fd);
out_free_linereq:
	linereq_free(lr);
	return ret;
}
```

先检查用户结构、线路数、保留字段和配置合法性，再分配linereq及柔性数组、取得gdev引用。为所有line写req回指并初始化delayed work，保证以后任何部分成功的清理都可安全调用停止路径。label分配失败发生得更早，此时尚无desc，不会尝试停止未建立的线路事件。

初始化mutex、wait与FIFO元数据以后，逐条执行S3/S4。`gpiod_request_user()` 成功就立即把desc存入数组，随后transitory、方向或事件配置失败也能被统一清理找到。配置flags先进入desc，输出方向函数可能已经改变硬件；失败归还desc不等于恢复这些变化。输入事件处理者注册以前，req、wait等依赖已准备好。

默认容量取num_lines×16，非零大值在此版本截到64×16；实际kfifo分配还按其容量规则处理，所以UAPI把输入称为建议量。每条线的REQUESTED通知发生在循环内，观察者可能看到早期线路已请求、随后整体失败再释放；接口没有向外提供原子硬件事务。

最后建立注销通知、预留fd号并创建匿名文件。匿名文件的private_data接管lr；若copy_to_user失败，fput会触发release，所以该分支直接返回，不再跳到linereq_free。只有参数复制成功才fd_install。这里分开了“预留一个号码”“文件持有对象”“把文件装入进程表”，避免把任一中间状态说成用户已取得可用请求。

## 6.3\_最后文件引用消失时怎样归还整组线路

line_fileops.release在file最终释放时执行，不是每次关闭任意一个dup副本都执行。它调用统一清理函数：

```c
/**
 * @brief 文件最后释放时回收private_data中的线路请求。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static int linereq_release(struct inode *inode, struct file *file)
{
	struct linereq *lr = file->private_data;

	linereq_free(lr);
	return 0;
}
```

```c
/**
 * @brief 按数组正向停止并释放已记录的线路，最后回收共享队列与引用。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static void linereq_free(struct linereq *lr)
{
	struct line *line;
	unsigned int i;

	if (lr->device_unregistered_nb.notifier_call)
		blocking_notifier_chain_unregister(&lr->gdev->device_notifier,
						   &lr->device_unregistered_nb);

	for (i = 0; i < lr->num_lines; i++) {
		line = &lr->lines[i];
		if (!line->desc)
			continue;

		edge_detector_stop(line);
		if (line_has_supinfo(line))
			supinfo_erase(line);
		gpiod_free(line->desc);
	}
	kfifo_free(&lr->events);
	kfree(lr->label);
	gpio_device_put(lr->gdev);
	kvfree(lr);
}
```

非空desc标记该项确实取得过请求。循环从0递增，先edge_detector_stop，再移除需要的附加信息，最后gpiod_free。停止事件必须早于释放desc和lr，因为IRQ线程及去抖work会沿line读取desc、沿req访问队列。停止函数的唯一展开在 [事件退出](P07_事件缓冲与读取.md#7.5_为什么必须先停止事件再回收请求)。

所有线路处理后才释放FIFO、label、gdev引用和lr。这个次序既适用于创建部分失败，也适用于成功文件退出；判断依据是已记录的状态，而不是假定每根线都完整初始化。gpiod_free的公共实现复用 [描述符归还](P02_描述符请求与托管释放.md#2.4_释放撤销占用而不是恢复寄存器快照)，不在这里重复函数体。

文件操作持有引用与事件生产者停止是两类保护：前者避免请求正在被read/ioctl使用时提前最终释放，后者处理不靠用户fd调用进入的IRQ/work。对象引用也不能代替业务停机。需要先断言reset、关闭enable的设备，必须在还持有请求时执行自己的协议。

## 6.4\_配置互斥为什么不是多线事务回滚

已经交付的request可通过SET_CONFIG改变方向等配置。互斥锁保证一个请求的配置步骤不被同类配置穿插，并不为每根线保存旧硬件状态。

```c
/**
 * @brief 锁内逐线应用配置；中途错误直接返回，已改变成员不自动复原。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static long linereq_set_config(struct linereq *lr, void __user *ip)
{
	struct gpio_v2_line_config lc;
	struct gpio_desc *desc;
	struct line *line;
	unsigned int i;
	u64 flags, edflags;
	int ret;

	if (copy_from_user(&lc, ip, sizeof(lc)))
		return -EFAULT;

	ret = gpio_v2_line_config_validate(&lc, lr->num_lines);
	if (ret)
		return ret;

	guard(mutex)(&lr->config_mutex);

	for (i = 0; i < lr->num_lines; i++) {
		line = &lr->lines[i];
		desc = lr->lines[i].desc;
		flags = gpio_v2_line_config_flags(&lc, i);
		/*
		 * Lines not explicitly reconfigured as input or output
		 * are left unchanged.
		 */
		if (!(flags & GPIO_V2_LINE_DIRECTION_FLAGS))
			continue;
		gpio_v2_line_config_flags_to_desc_flags(flags, &desc->flags);
		edflags = flags & GPIO_V2_LINE_EDGE_DETECTOR_FLAGS;
		if (flags & GPIO_V2_LINE_FLAG_OUTPUT) {
			int val = gpio_v2_line_config_output_value(&lc, i);

			edge_detector_stop(line);
			ret = gpiod_direction_output(desc, val);
			if (ret)
				return ret;
		} else {
			ret = gpiod_direction_input(desc);
			if (ret)
				return ret;

			ret = edge_detector_update(line, &lc, i, edflags);
			if (ret)
				return ret;
		}

		WRITE_ONCE(line->edflags, edflags);

		gpiod_line_state_notify(desc, GPIO_V2_LINE_CHANGED_CONFIG);
	}
	return 0;
}
```

copy与合法性检查在锁之前，随后guard持有config_mutex直到返回。每线如果没有明确INPUT/OUTPUT就跳过；否则先把flags写到desc。输出路径先停止旧事件，再设置方向和初值；输入路径先设置方向，再更新事件检测。任一步失败直接返回，因此此前线路已应用的配置、当前线路已执行的部分操作都不能假定撤销。

成功后才写edflags并发CONFIG通知。事件处理者读取的edflags允许实现所述的短暂旧值，config_mutex不是阻塞全部IRQ的锁。失败处理应依据设备安全协议重新建立明确状态，不能以“ioctl返回失败”证明旧配置完整保留。练习可设想先改变offset0、再让offset2的事件设置失败：检查返回路径能找到哪一处恢复offset0。答案是本函数没有这样的回滚；这正是调用者必须理解的边界。

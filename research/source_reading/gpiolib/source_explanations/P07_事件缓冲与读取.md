# 第7章\_事件缓冲与读取

请求已经拥有线路以后，事件仍可能在用户进程被调度前发生多次。本页把IRQ回调、每线状态、共享FIFO和用户read接成一条路径，解释有界缓冲与等待各解决什么。先读 [用户请求模块导读](../navigation/P05_用户请求怎样持有线路与事件.md#5.3_一条事件怎样从IRQ状态进入进程)，版本与配置见 [总索引](../navigation/P01_Linux_6.12_GPIO源码阅读索引.md#1.1_版本身份与阅读路线)。

摘录来自NXP Linux 6.12.20固定提交 `dfaf2136deb2af2e60b994421281ba42f1c087e0` 的 `drivers/gpio/gpiolib-cdev.c`，保留GPL-2.0许可。中文Doxygen为仓库补充；这里展开无硬件授时、无软件去抖的普通边沿主路径，同时指出分支入口，不把它当成所有控制器唯一实现。

## 7.1\_建立事件检测时先准备哪些共享状态

请求创建已经初始化line.req和共享wait，检测设置还要分配FIFO、选择边沿与时钟能力、建立IRQ处理者。FIFO在整个请求共享，第二根事件线不能把第一根正在使用的队列重新初始化。

```c
/**
 * @brief 已有FIFO直接复用，首次需要事件时才分配存储。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static int edge_detector_fifo_init(struct linereq *req)
{
	if (kfifo_initialized(&req->events))
		return 0;

	return kfifo_alloc(&req->events, req->event_buffer_size, GFP_KERNEL);
}
```

```c
/**
 * @brief 先建立缓冲与去抖分支，再把逻辑边沿转换为IRQ触发并安装处理者。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static int edge_detector_setup(struct line *line,
			       struct gpio_v2_line_config *lc,
			       unsigned int line_idx, u64 edflags)
{
	u32 debounce_period_us;
	unsigned long irqflags = 0;
	u64 eflags;
	int irq, ret;
	char *label;

	eflags = edflags & GPIO_V2_LINE_EDGE_FLAGS;
	if (eflags) {
		ret = edge_detector_fifo_init(line->req);
		if (ret)
			return ret;
	}
	if (gpio_v2_line_config_debounced(lc, line_idx)) {
		debounce_period_us = gpio_v2_line_config_debounce_period(lc, line_idx);
		ret = debounce_setup(line, debounce_period_us);
		if (ret)
			return ret;
		line_set_debounce_period(line, debounce_period_us);
	}

	/* detection disabled or sw debouncer will provide edge detection */
	if (!eflags || READ_ONCE(line->sw_debounced))
		return 0;

	if (IS_ENABLED(CONFIG_HTE) &&
	    (edflags & GPIO_V2_LINE_FLAG_EVENT_CLOCK_HTE))
		return hte_edge_setup(line, edflags);

	irq = gpiod_to_irq(line->desc);
	if (irq < 0)
		return -ENXIO;

	if (eflags & GPIO_V2_LINE_FLAG_EDGE_RISING)
		irqflags |= test_bit(FLAG_ACTIVE_LOW, &line->desc->flags) ?
			IRQF_TRIGGER_FALLING : IRQF_TRIGGER_RISING;
	if (eflags & GPIO_V2_LINE_FLAG_EDGE_FALLING)
		irqflags |= test_bit(FLAG_ACTIVE_LOW, &line->desc->flags) ?
			IRQF_TRIGGER_RISING : IRQF_TRIGGER_FALLING;
	irqflags |= IRQF_ONESHOT;

	label = make_irq_label(line->req->label);
	if (IS_ERR(label))
		return PTR_ERR(label);

	/* Request a thread to read the events */
	ret = request_threaded_irq(irq, edge_irq_handler, edge_irq_thread,
				   irqflags, label, line);
	if (ret) {
		free_irq_label(label);
		return ret;
	}

	line->irq = irq;
	return 0;
}
```

eflags提取逻辑边沿需求，有边沿才调用FIFO分配。去抖配置可能建立软件工作路径；没有边沿或软件去抖已经负责检测时，普通IRQ设置在此返回。HTE分支另走硬件授时设置，不能从普通线程函数推断它的所有保证。

普通路径调用gpiod_to_irq，负值在这里转成ENXIO。RISING/FALLING依据desc的ACTIVE_LOW互换物理触发标志，ONESHOT配合线程处理，最后注册主处理函数与线程。只有注册成功才写line.irq；失败释放临时label。input方向由调用方先建立，to_irq本身不负责替它切方向。

FIFO、line状态、IRQ action分属不同载体：FIFO保存尚未读走的事件，line保存该输入检测的运行状态，IRQ action把通知连到函数。它们都不保存输入变化之前全部物理历史。

## 7.2\_时间与边沿怎样变成一条记录

主处理者尽早记录时间，避免全部时间误差都推迟到线程获得CPU以后；但“尽早”仍然包含输入检测和中断传递延迟。

```c
/**
 * @brief 在普通主处理路径保存软件时间戳及多线请求序号，再唤起线程。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static irqreturn_t edge_irq_handler(int irq, void *p)
{
	struct line *line = p;
	struct linereq *lr = line->req;

	/*
	 * Just store the timestamp in hardirq context so we get it as
	 * close in time as possible to the actual event.
	 */
	line->timestamp_ns = line_event_timestamp(line);

	if (lr->num_lines != 1)
		line->req_seqno = atomic_inc_return(&lr->seqno);

	return IRQ_WAKE_THREAD;
}
```

```c
/**
 * @brief 补齐事件字段并提交共享队列；nested线程可能在此才取得时间。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static irqreturn_t edge_irq_thread(int irq, void *p)
{
	struct line *line = p;
	struct linereq *lr = line->req;
	struct gpio_v2_line_event le;

	/* Do not leak kernel stack to userspace */
	memset(&le, 0, sizeof(le));

	if (line->timestamp_ns) {
		le.timestamp_ns = line->timestamp_ns;
	} else {
		/*
		 * We may be running from a nested threaded interrupt in
		 * which case we didn't get the timestamp from
		 * edge_irq_handler().
		 */
		le.timestamp_ns = line_event_timestamp(line);
		if (lr->num_lines != 1)
			line->req_seqno = atomic_inc_return(&lr->seqno);
	}
	line->timestamp_ns = 0;

	switch (READ_ONCE(line->edflags) & GPIO_V2_LINE_EDGE_FLAGS) {
	case GPIO_V2_LINE_FLAG_EDGE_BOTH:
		le.id = line_event_id(gpiod_get_value_cansleep(line->desc));
		break;
	case GPIO_V2_LINE_FLAG_EDGE_RISING:
		le.id = GPIO_V2_LINE_EVENT_RISING_EDGE;
		break;
	case GPIO_V2_LINE_FLAG_EDGE_FALLING:
		le.id = GPIO_V2_LINE_EVENT_FALLING_EDGE;
		break;
	default:
		return IRQ_NONE;
	}
	line->line_seqno++;
	le.line_seqno = line->line_seqno;
	le.seqno = (lr->num_lines == 1) ? le.line_seqno : line->req_seqno;
	le.offset = gpio_chip_hwgpio(line->desc);

	linereq_put_event(lr, &le);

	return IRQ_HANDLED;
}
```

line_event_timestamp默认读单调时钟，REALTIME选项读实时时钟，HTE有单独语义。线程先清零整个事件结构，避免未初始化padding泄露。已有timestamp就采用主处理者记录；没有则按nested线程场景补记，多线请求的seqno也在该分支补取，随后清掉缓存时间戳。

双边沿分支读取当前逻辑GPIO值，line_event_id把非零映射为RISING、0映射为FALLING；单边沿则由配置直接选择id。此固定双边沿代码没有单独检查gpiod_get_value_cansleep的负错误值，不能把它讲成“所有采样错误都会返回给用户”或精确边沿锁存。对快速变化和总线读取错误必须结合Provider核对；本文记录实现边界，不在只读源码任务中改上游代码。

line_seqno在该线处理路径递增；多线req_seqno由请求原子计数取得，单线请求直接复用line_seqno。offset经gpio_chip_hwgpio得到chip内编号。多线并发时序号分配时间和真正入队时间不必重合，所以不能无条件把共享队列排序等同于物理边沿严格顺序。输入前端已经遗漏的变化没有生成这些编号，序号连续也证明不了前端无损。

## 7.3\_满队列怎样改变可观察的历史

事件生产者不能无限等待用户进程腾空间，否则用户调度延迟会反过来阻塞整个IRQ处理链。这里选择有界队列满时覆盖最旧记录，代价是旧历史丢失。

```c
/**
 * @brief 用等待队列锁保护FIFO；满时先跳过最旧项再插入新事件。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static void linereq_put_event(struct linereq *lr,
			      struct gpio_v2_line_event *le)
{
	bool overflow = false;

	scoped_guard(spinlock, &lr->wait.lock) {
		if (kfifo_is_full(&lr->events)) {
			overflow = true;
			kfifo_skip(&lr->events);
		}
		kfifo_in(&lr->events, le, 1);
	}
	if (!overflow)
		wake_up_poll(&lr->wait, EPOLLIN);
	else
		pr_debug_ratelimited("event FIFO is full - event dropped\n");
}
```

满队列分支先kfifo_skip，再kfifo_in，新事件被保留。非overflow路径才wake_up_poll；overflow时已有未读数据，队列早已可读，代码只输出限速debug信息。该信息不是每次都交付给用户的错误记录，应用应结合序号、读取者数量和自己的业务协议判断是否可继续。

这把wait.lock同时用于read取出，因此不会由两个普通读者同时取得同一个队首事件；但dup共享一个FIFO，读者之间会瓜分记录。需要广播时应由一个读者取出再在用户空间明确分发，不靠复制fd实现。

## 7.4\_可读通知为何不能替代取出记录

poll把当前线程的等待关系挂到request的waitqueue，再检查FIFO。read真正从FIFO搬走记录，两者间允许其他读者竞争。

```c
/**
 * @brief 登记等待并在队列锁下检查可读性，Provider已撤销时报告错误与挂断。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static __poll_t linereq_poll(struct file *file,
			     struct poll_table_struct *wait)
{
	struct linereq *lr = file->private_data;
	__poll_t events = 0;

	guard(srcu)(&lr->gdev->srcu);

	if (!rcu_access_pointer(lr->gdev->chip))
		return EPOLLHUP | EPOLLERR;

	poll_wait(file, &lr->wait, wait);

	if (!kfifo_is_empty_spinlocked_noirqsave(&lr->events,
						 &lr->wait.lock))
		events = EPOLLIN | EPOLLRDNORM;

	return events;
}
```

```c
/**
 * @brief 锁内等待并取出完整事件，解锁后复制用户缓冲；非阻塞空队列返回EAGAIN。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static ssize_t linereq_read(struct file *file, char __user *buf,
			    size_t count, loff_t *f_ps)
{
	struct linereq *lr = file->private_data;
	struct gpio_v2_line_event le;
	ssize_t bytes_read = 0;
	int ret;

	guard(srcu)(&lr->gdev->srcu);

	if (!rcu_access_pointer(lr->gdev->chip))
		return -ENODEV;

	if (count < sizeof(le))
		return -EINVAL;

	do {
		scoped_guard(spinlock, &lr->wait.lock) {
			if (kfifo_is_empty(&lr->events)) {
				if (bytes_read)
					return bytes_read;

				if (file->f_flags & O_NONBLOCK)
					return -EAGAIN;

				ret = wait_event_interruptible_locked(lr->wait,
						!kfifo_is_empty(&lr->events));
				if (ret)
					return ret;
			}

			if (kfifo_out(&lr->events, &le, 1) != 1) {
				/*
				 * This should never happen - we hold the
				 * lock from the moment we learned the fifo
				 * is no longer empty until now.
				 */
				WARN(1, "failed to read from non-empty kfifo");
				return -EIO;
			}
		}

		if (copy_to_user(buf + bytes_read, &le, sizeof(le)))
			return -EFAULT;
		bytes_read += sizeof(le);
	} while (count >= bytes_read + sizeof(le));

	return bytes_read;
}
```

两者先在gdev的SRCU保护中核对chip是否仍可用。poll已不可用时返回HUP|ERR；read入口已不可用时返回ENODEV。这些是入口防御，不能扩大成“任意时刻强拆Provider都能让所有阻塞调用安全立即返回”；正常顺序仍应先退出请求者。

read要求缓冲至少容纳一个完整事件。锁内发现空队列时，若已经复制过若干事件就返回已有字节；没有数据且非阻塞则EAGAIN，否则通过wait_event_interruptible_locked放下锁等待，并在条件满足后重新持锁。取出与判空在同一锁区，出锁后才copy_to_user，避免在自旋锁内执行可能发生缺页的复制。

复制失败返回EFAULT，此时该事件已出队；不能通过重试保证拿回原记录。一次read可以返回多个完整记录，也可能少于请求缓冲容量，调用者按实际字节数解析。唤醒只让任务有机会继续运行，不携带队首记录，也不保证它马上得到CPU。主线程序的非阻塞read在poll以后仍处理EAGAIN，正是为这个交接保留正确分支。

## 7.5\_为什么必须先停止事件再回收请求

file最终释放会进入linereq_free，但IRQ线程和去抖work通过line.req访问队列，不是通过用户fd进入。不能先free FIFO再期待关闭文件阻止这些生产者。

```c
/**
 * @brief 释放并等待IRQ、停止授时分支及延迟工作，再清检测标志。
 * @note 仓库补充中文阅读说明；固定版本源码，非独立可编译程序。
 */
static void edge_detector_stop(struct line *line)
{
	if (line->irq) {
		free_irq_label(free_irq(line->irq, line));
		line->irq = 0;
	}

#ifdef CONFIG_HTE
	if (READ_ONCE(line->edflags) & GPIO_V2_LINE_FLAG_EVENT_CLOCK_HTE)
		hte_ts_put(&line->hdesc);
#endif

	cancel_delayed_work_sync(&line->work);
	WRITE_ONCE(line->sw_debounced, 0);
	WRITE_ONCE(line->edflags, 0);
	line_set_debounce_period(line, 0);
	/* do not change line->level - see comment in debounced_value() */
}
```

有IRQ时先free_irq并释放其label，清irq号码；HTE分支归还授时注册；cancel_delayed_work_sync等待去抖工作退出，再清sw_debounced、edflags和去抖周期。line.level特意保持，避免正在按既有规则读取去抖值的路径突然失去其所依赖的值。

调用方随后才gpiod_free并最终回收整个请求，完整顺序见 [最后引用释放](P06_用户请求的发布与回收.md#6.3_最后文件引用消失时怎样归还整组线路)。这处理的是内核事件状态，不替用户执行“关闭供电”等业务协议。当前页未执行IRQ运行或用户事件压力实验，读者可从 [主线完整实验](../../../../knowledge/driver_model/gpio/P08_用户空间_ABI_迁移与方案选择.md#8.5_在模拟器中准备一组受控的线路) 观察受控路径；不能把两条慢速事件成功外推为任意频率无损。
